Create a SaltMinionKey in namespace "{{ .Values.namespace }}":

  kind: SaltMinionKey
  metadata:
    name: <minion-id>
    namespace: {{ .Values.namespace }}
  spec:
    minionId: <minion-id>
    publicKey: |
      -----BEGIN PUBLIC KEY-----
      ...
      -----END PUBLIC KEY-----
