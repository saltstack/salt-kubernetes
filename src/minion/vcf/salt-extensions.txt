# Salt extension packages installed into the Salt onedir Python environment
# when the image is built (src/minion/vcf/Dockerfile).
# Add/remove/pin any pip-installable Salt extension here. Salt itself is
# constrained to the installed onedir version during this step, so an extension's
# "salt" requirement never replaces it. See ../kubernetes/salt-extensions.txt for the
# Kubernetes image's set.

# Kubernetes execution/state modules and cluster-API access, used for
# kube-bench based compliance assessments.
# Source: https://github.com/salt-extensions/saltext-kubernetes
saltext.kubernetes==2.1.0

# Vault integration (ext_pillar, sdb driver, execution modules) - lets
# pillar values reference sdb://vault_sdb/... instead of plaintext.
# Source: https://github.com/salt-extensions/saltext-vault
saltext.vault==1.8.0

# Salt extension for VMware Cloud Foundation 9.x. Targets the full VCF stack:
# vCenter, NSX, SDDC Manager, VCF Operations, VKS (Supervisor), VMSP (mediated),
# vSAN, and standalone ESXi.
saltext.vcf==1.0.0

# A Salt extension for out-of-band management of bare-metal servers via their 
# BMC (Baseboard Management Controller). Speaks Redfish first and IPMI as a 
# fallback (auto-detected by default),
saltext.bmc==0.0.1
